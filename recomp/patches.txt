# Instruction patches for game-specific problems: one per line, each with a comment
# saying what it fixes. See gcn-recomp's README for the format.

# The game's logic rate (issues #12, #59). The game keeps its frames per second in one
# variable, sda-30252 (0x80385AF4: 60 in flight, 30 on foot and in some cutscenes), and
# converts by it wherever time meets frames: its time step, derived every frame in
# fn_8003AEB0, the frame budget the main loop paces itself by, and some 80 sites in main.dol
# and the RELs that turn a duration or a rate into frames. These are all the game's stores
# to it: r13-30252 in main.dol, 0x80380000 + 23284 in the RELs, or a register holding that
# address. Each stores what src/frame_rate.cpp makes of the value instead: the value
# divided by the step scale, so twice it at 120 fps and the game's own value at 60. Every
# reader then follows. A store missed here is caught at the next frame and logged.
8000E13C { uint32_t sfassault_fps_store(uint32_t); ST32((c->r[13] + (uint32_t)-30252), sfassault_fps_store(c->r[6])); }  # main: 60 at boot
8003AFEC { uint32_t sfassault_fps_store(uint32_t); ST32((c->r[13] + (uint32_t)-30252), sfassault_fps_store(c->r[0])); }  # main: 30
80057824 { uint32_t sfassault_fps_store(uint32_t); ST32((c->r[13] + (uint32_t)-30252), sfassault_fps_store(c->r[0])); }  # main: 60
800AB8C4 { uint32_t sfassault_fps_store(uint32_t); ST32((c->r[13] + (uint32_t)-30252), sfassault_fps_store(c->r[0])); }  # main: 60 / n
800C2E90 { uint32_t sfassault_fps_store(uint32_t); ST32((c->r[13] + (uint32_t)-30252), sfassault_fps_store(c->r[0])); }  # main: 60
81810DD4 { uint32_t sfassault_fps_store(uint32_t); ST32((c->r[4] + (uint32_t)23284), sfassault_fps_store(c->r[5])); }  # m2.rel, the menus
81831318 { uint32_t sfassault_fps_store(uint32_t); ST32((c->r[3] + (uint32_t)23284), sfassault_fps_store(c->r[0])); }  # m3.rel
818858C8 { uint32_t sfassault_fps_store(uint32_t); ST32((c->r[3] + (uint32_t)23284), sfassault_fps_store(c->r[0])); }  # m5.rel, the briefing: 30
81888040 { uint32_t sfassault_fps_store(uint32_t); ST32((c->r[4] + (uint32_t)23284), sfassault_fps_store(c->r[0])); }  # m6.rel: 30
818CC9C4 { uint32_t sfassault_fps_store(uint32_t); ST32((c->r[6] + (uint32_t)23284), sfassault_fps_store(c->r[7])); }  # m9.rel
818008AC { uint32_t sfassault_fps_store(uint32_t); ST32(c->r[4], sfassault_fps_store(c->r[7])); }  # m0.rel: 30, through r4

818CCCB4 { uint32_t sfassault_fps_store(uint32_t); ST32(c->r[4], sfassault_fps_store(c->r[0])); }  # m9.rel, through r4

# The game's retrace callback (fn_8000DE20) counts in 60 Hz retraces: a retrace counter
# (sda-30272), a 16.16 clock (sda-30260, 1092 = 1/60 s), the frame budget the main loop
# waits on (sda-30176: a frame adds 1/fps, each retrace pays 1/60 s back, down to -1/60 s),
# and its frame meter (sda-30168, a second of 60 retraces). At 120 fps the VI retraces twice
# a field, and a 120 Hz retrace paid for a 1/60 s frame: what runs at 30 in the original (on
# foot, the boot logos) ran at 120 on a 60 time step, twice its speed (issue #59). These make
# a retrace count for its share of a field (src/frame_rate.cpp): 546 at 120 fps, 1092 at 60,
# and the counts taken on every second retrace at 120.
8000DE20 { void sfassault_retrace(void); sfassault_retrace(); c->r[5] = LD32((c->r[13] + (uint32_t)-30272)); }
8000DE2C { uint32_t sfassault_retrace_whole(void); c->r[5] = c->r[5] + sfassault_retrace_whole(); }
8000DE30 { uint32_t sfassault_retrace_ticks(void); c->r[0] = c->r[4] + sfassault_retrace_ticks(); }
8000DE38 { uint32_t sfassault_retrace_ticks(void); c->cr[0] = CMPS((int32_t)c->r[3], -(int32_t)sfassault_retrace_ticks()); }
8000DE44 { uint32_t sfassault_retrace_ticks(void); c->r[0] = c->r[3] - sfassault_retrace_ticks(); }
8000DE50 { uint32_t sfassault_retrace_whole(void); c->r[0] = c->r[3] + sfassault_retrace_whole(); }

# Skip the intro (issue #41, src/skip_intro.cpp). The logo task's step from its state 1 to
# 2 (once its file is loaded) goes to its last state, 15, with the setting on.
800111C0 { extern uint32_t sfa_logos_next_state(uint32_t); ST32((c->r[31] + (uint32_t)52), sfa_logos_next_state(c->r[0])); }
# The flow's state 3 starts a movie: the opening one the first time after boot, which the
# setting can leave out (the flow then loads the title at once).
8000F19C { extern int sfa_skip_boot_movie(void); if (!sfa_skip_boot_movie()) { c->lr = 0x8000F1A0u; fn_800C2E18(c); } }

# Skip cutscenes at once (issue #41, src/cutscenes.cpp). With the setting on, a Start press
# a cutscene can't act on yet is kept until it can. The movie task, every frame it runs:
800C327C { extern void sfa_movie_frame(uint32_t, uint32_t); sfa_movie_frame(c->r[3], LD32(0x8029A8C0u)); c->r[0] = LD32((c->r[3] + (uint32_t)84)); }
# where it tests Start,
800C3374 { extern uint32_t sfa_movie_start(uint32_t, uint32_t); c->r[0] = sfa_movie_start(c->r[31], c->r[0] & 0x1000u); SET_CR0(c->r[0]); }
# and how many frames it must have played first: 30, or 1 with the setting on.
800C3398 { extern int32_t sfa_movie_skip_after(void); c->cr[0] = CMPS((int32_t)c->r[0], sfa_movie_skip_after()); }
# A scene the game plays itself, where it tests Start each frame.
